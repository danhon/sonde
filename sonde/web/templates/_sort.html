{#- Shared sorting macros. Extracted from followers.html so every table sorts the same way and carries the same filters — the /groups table was built without them and could not sort at all. -#} {%- macro qs(base, overrides={}) -%} {%- set params = {} -%} {%- for key, value in base.items() -%}{%- set _ = params.update({key: value}) -%}{%- endfor -%} {%- set _ = params.update(overrides) -%} {%- set parts = [] -%} {%- for key, value in params.items() -%} {%- if value not in (None, '', False) -%} {%- set _ = parts.append(key ~ '=' ~ (value | string | urlencode)) -%} {%- endif -%} {%- endfor -%} ?{{ parts | join('&') }} {%- endmacro -%} {%- macro th(base, key, label, align='left') -%} {%- set active = (base.get('order') == key) -%} {%- set flip = 'asc' if (active and base.get('direction', 'desc') == 'desc') else 'desc' -%} {{ label }} {{ '↑' if base.get('direction') == 'asc' else '↓' }} {%- endmacro -%}
